async function loadReport() { const id = document.getElementById("studentId").value.trim(); const section = sessionStorage.getItem("section-role") || "male"; // male/female const reportDiv = document.getElementById("report"); const errorDiv = document.getElementById("error"); const printBtn = document.getElementById("printBtn"); reportDiv.innerHTML = ""; errorDiv.textContent = ""; printBtn.style.display = "none"; if (!id) { errorDiv.textContent = "❌ الرجاء إدخال رقم الهوية."; return; } try { const res = await fetch(`/api/report/${id}?section=${section}`); if (!res.ok) { if (res.status === 404) { errorDiv.textContent = "❌ الطالب غير موجود."; } else { errorDiv.textContent = "❌ خطأ في تحميل التقرير."; } return; } const data = await res.json(); // عرض بيانات الطالب كما كان سابقاً let html = `

مجمع زايد التعليمي - الظيت

الاسم: ${data.student["الاسم"]}

الشعبة: ${data.student["الشعبة"]}

`; let notes = []; data.subjects.forEach(sub => { html += ` `; if (sub.note && sub.note.trim() && sub.note !== "-") { notes.push(`${sub.name}: ${sub.note}`); } }); html += "
المادة تكويني مشاركة مهمة التزام
${sub.name} ${sub.formative} ${sub.participation} ${sub.task} ${sub.commitment}
"; if (notes.length > 0) { html += `

📝 الملاحظات:

    ${notes.map(n => `
  • ${n}
    • `).join("")}
`; } reportDiv.innerHTML = html; printBtn.style.display = "inline-block"; } catch (err) { console.error("Error loading report:", err); errorDiv.textContent = "❌ حدث خطأ غير متوقع."; } }
